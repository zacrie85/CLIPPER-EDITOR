    24→/* ============================================================
    25→   v1.2.6 — ANTI-FREEZE EKSPOR
    26→   Laporan: "Encoding video 270/7239… 1%" beku + job lain mati di
    27→   "Menunggu antrean… 0%" selamanya + tombol BATAL tidak bereaksi.
    28→
    29→   Akar masalah yang diperbaiki di file ini:
    30→   1. await seekVideo() bisa menggantung selamanya (rAF mati saat
    31→      window minimize / event seeked hilang) → kini dilindungi
    32→      raceStall(): batas waktu 45 dtk + BATAL tetap bereaksi.
    33→   2. Tunggu onloadedmetadata video tanpa batas → kini openVideo()
    34→      dengan timeout + pesan jelas.
    35→   3. Loop backpressure encoder bisa berputar selamanya bila encoder
    36→      mati → kini dibatasi 30 dtk + cek encodeErr/state tiap putaran.
    37→   4. EKSPOR PARTS menumpuk SEMUA blob di RAM dulu (20 bagian ×
    38→      ratusan MB = OOM crash) → kini tiap bagian DIUNDUH SEGERA.
    39→   5. EKSPOR PARTS decode audio penuh 48 kHz BERULANG per bagian
    40→      (20× decode!) → kini dibaca SEKALI untuk semua bagian.
    41→   6. BATCH SEMUA hanya memakai media[0] untuk semua range →
    42→      output video ke-2 dst SALAH → kini tiap video diproses sendiri.
    43→   7. GABUNG VIDEO: timestamp audio mulai dari 0 per sumber →
    44→      audio saling tumpang → kini offset global berjalan.

    47→/** lindungi promise dari menggantung selamanya: batas waktu + BATAL aktif.
    48→    isCancelled dicek tiap 250 ms — tombol BATALKAN bekerja bahkan saat
    49→    seek/encode sedang macet. */
    50→function raceStall<T>(
    51→  p: Promise<T>,
    52→  isCancelled: (() => boolean) | undefined,
    53→  ms: number,
    54→  err: Error
    55→): Promise<T> {
    56→  return new Promise<T>((resolve, reject) => {
    57→    let done = false;
    58→    let timer: ReturnType<typeof setTimeout> | null = null;
    59→    let iv: ReturnType<typeof setInterval> | null = null;
    60→    const cleanup = () => {
    61→      if (timer) { clearTimeout(timer); timer = null; }
    62→      if (iv) { clearInterval(iv); iv = null; }
    63→    };
    64→    if (isCancelled) {
    65→      iv = setInterval(() => {
    66→        if (!done && isCancelled()) {
    67→          done = true;
    68→          cleanup();
    69→          reject(new Error("Dibatalkan"));
    70→        }
    71→      }, 250);
    72→    }
    73→    timer = setTimeout(() => {
    74→      if (done) return;
    75→      done = true;
    76→      cleanup();
    77→      reject(err);
    78→    }, ms);
    79→    p.then(
    80→      (v) => {
    81→        if (done) return;
    82→        done = true;
    83→        cleanup();
    84→        resolve(v);
    85→      },
    86→      (e) => {
    87→        if (done) return;
    88→        done = true;
    89→        cleanup();
    90→        reject(e instanceof Error ? e : new Error(String(e)));
    91→      }
    92→    );
    93→  });
    94→}

   114→export function outputDims(settings: ExportSettings, srcW: number, srcH: number): { w: number; h: number } {
   115→  const even = (n: number) => Math.max(2, Math.round(n / 2) * 2);
   116→  if (settings.autoMode) {
   117→    const k = settings.scale / 100;
   118→    const dims = even(srcW * k), evenh = even(srcH * k);
   119→    // rasio mengikuti sumber; vertikal dibalik sesuai konvensi
   120→    return { w: dims, h: evenh };
   121→  }
   122→  // v1.2.7 — guard: resolusi tak dikenal (mis. localStorage korup) jangan
   123→  // sampai TypeError merobohkan seluruh UI — jatuh ke 720p.
   124→  const r = RESOLUTIONS[settings.resolution] ?? RESOLUTIONS["720p"];
   125→  const d = settings.aspect === "9:16" ? r.portrait : r.landscape;
   126→  return { w: d.w, h: d.h };
   127→}

   129→export function videoBitrate(settings: ExportSettings, duration: number, srcW: number, srcH: number): number {
   130→  const q = settings.quality === "high" ? 1 : settings.quality === "balanced" ? 0.65 : 0.42;
   131→  let mbps: number;
   132→  if (settings.autoMode) {
   133→    const dims = outputDims(settings, srcW, srcH);
   134→    // skala bitrate terhadap piksel relatif ke 1080p
   135→    const px = (dims.w * dims.h) / (1920 * 1080);
   136→    mbps = Math.max(2, 10 * Math.sqrt(px));
   137→  } else {
   138→    const r = RESOLUTIONS[settings.resolution];
   139→    mbps = settings.fps >= 60 ? r.mbps60 : r.mbps30;
   140→  }
   141→  let bitrate = Math.round(mbps * 1e6 * q);
   142→
   143→  // KOMPRES PINTAR — target MB diutamakan bila diisi
   144→  if (settings.targetMB && settings.targetMB > 0 && duration > 0) {
   145→    const totalBits = settings.targetMB * 8 * 1048576;
   146→    const audioBits = 160_000 * duration;
   147→    const vb = Math.max(300_000, Math.floor(totalBits - audioBits) / duration);
   148→    bitrate = Math.min(bitrate, Math.round(vb));
   149→  }
   150→  return bitrate;
   151→}

   153→/** codec string H.264 dengan level otomatis (5.1/5.2 utk 2K/4K incl 60fps) */
   154→export function h264Codec(settings: ExportSettings, w: number, h: number): string {
   155→  if (settings.autoMode) {
   156→    const px = w * h;
   157→    if (px >= 1920 * 2160) return settings.fps >= 60 ? "avc1.640034" : "avc1.640033";
   158→    if (px >= 1280 * 1440) return "avc1.640033";
   159→    if (px >= 1280 * 720) return "avc1.64002a";
   160→    return "avc1.640028";
   161→  }
   162→  const r = RESOLUTIONS[settings.resolution];
   163→  if (r.key === "4k") return settings.fps >= 60 ? "avc1.640034" : "avc1.640033";
   164→  if (r.key === "2k") return "avc1.640033";
   165→  if (r.key === "1080p") return "avc1.64002a";
   166→  return "avc1.640028";
   167→}

   190→async function assertBlobOk(blob: Blob, label: string): Promise<void> {
   191→  try {
   192→    await blob.slice(0, 64).arrayBuffer();
   193→    const tail = Math.max(0, blob.size - 65_536);
   194→    await blob.slice(tail).arrayBuffer();
   195→  } catch {
   196→    throw new Error(
   197→      `${label}: hasil encode tidak dapat dibaca (memori penuh) — pilih SIMPAN KE FOLDER PILIHAN di panel Ekspor (ditulis langsung ke disk), atau tutup aplikasi lain lalu ulangi.`
   198→    );
   199→  }
   200→}

   219→export async function exportOne(opts: ExportOneOptions): Promise<Blob | null> {
   220→  const { media, clip, name } = opts;
   221→  const settings = clip.export;
   222→  const totalDur = opts.ranges.reduce((a, r) => a + (r.end - r.start), 0);
   223→  if (totalDur <= 0.05) throw new Error("Rentang ekspor kosong");
   224→
   225→  await ensureFontsLoaded();
   226→  opts.onProgress?.(0.01, "Menyiapkan encoder…");
   227→
   228→  const dims = outputDims(settings, media.width || 1920, media.height || 1080);
   229→  const bitrate = videoBitrate(settings, totalDur, media.width || 1920, media.height || 1080);
   230→  const codec = h264Codec(settings, dims.w, dims.h);
   231→  const fps = settings.fps;
   232→  const frameDurUs = Math.round(1e6 / fps);
   233→
   234→  const hw = settings.hw === "gpu" ? "prefer-hardware" : settings.hw === "cpu" ? "prefer-software" : "no-preference";
   235→
   236→  const videoCfg: VideoEncoderConfig = {
   237→    codec,
   238→    width: dims.w,
   239→    height: dims.h,
   240→    bitrate,
   241→    framerate: fps,
   242→    latencyMode: "quality",
   243→    hardwareAcceleration: hw as VideoEncoderConfig["hardwareAcceleration"],
   244→  };
   245→  const vsup = await VideoEncoder.isConfigSupported(videoCfg);
   246→  if (!vsup.supported) {
   247→    // fallback tanpa hw preference
   248→    const fallback = { ...videoCfg, hardwareAcceleration: "no-preference" as const };
   249→    const f2 = await VideoEncoder.isConfigSupported(fallback);
   250→    if (!f2.supported) throw new Error(`Encoder H.264 ${dims.w}×${dims.h} tidak didukung browser ini`);
   251→    Object.assign(videoCfg, fallback);
   252→  }
   253→
   254→  /* ----- audio ----- */
   255→  let audioSetup: AudioSetup | null = null;
   256→  let mixed: AudioBuffer | null = null;
   257→  let original: AudioBuffer | null = null;
   258→  if (opts.audioPre !== undefined) {
   259→    // v1.2.6 — sudah di-decode pemanggil (sekali untuk semua bagian split)
   260→    original = opts.audioPre;
   261→  } else {
   262→    try {
   263→      opts.onProgress?.(0.02, "Membaca audio video…");
   264→      original = media.type === "video" ? await decodeAudioFile(media.url) : null;
   265→    } catch {
   266→      original = null;
   267→    }
   268→  }
   269→
   270→  let musicBuf: AudioBuffer | null = null;
   271→  if (clip.music.preset !== "none") {
   272→    if (clip.music.customBuffer) {
   273→      musicBuf = clip.music.customBuffer;
   274→    } else {
   275→      opts.onProgress?.(0.03, "Mengomposisi musik AI…");
   276→      musicBuf = await composeMusic(clip.music.preset, totalDur, clip.music.fadeOut);
   277→    }
   278→  }
   279→
   280→  const hasOriginalAudio = !!original && media.type === "video";
   281→  const hasMusic = !!musicBuf;
   282→  if (hasOriginalAudio || hasMusic) {
   283→    opts.onProgress?.(0.05, "Mencampur audio…");
   284→    audioSetup = await pickAudioConfig(48000);
   285→    if (audioSetup) {
   286→      mixed = await renderMix({
   287→        duration: totalDur,
   288→        sampleRate: audioSetup.cfg.sampleRate,
   289→        channels: audioSetup.cfg.numberOfChannels,
   290→        original: hasOriginalAudio ? original : null,
   291→        originalStart: opts.ranges[0]?.start ?? 0,
   292→        // ducking: kecilkan audio asli saat musik dipakai
   293→        originalGain: hasMusic ? 1 - clip.music.duckOriginal * 0.8 : 1,
   294→        music: musicBuf,
   295→        musicGain: clip.music.gain,
   296→        musicLoop: true,
   297→        fadeIn: 0.4,
   298→        fadeOut: clip.music.fadeOut,
   299→      });
   300→    }
   301→  }
   302→
   303→  /* ----- muxer -----
   304→     v1.2.7 — fileHandle → tulis langsung ke disk (StreamTarget),
   305→     tanpa buffer raksasa di RAM (menyembuhkan blob 0 byte di mesin
   306→     RAM kecil). fastStart false = moov ditulis saat finalize. */
   307→  let writable: FileSystemWritableFileStream | null = null;
   308→  let target: ArrayBufferTarget | FileSystemWritableFileStreamTarget = new ArrayBufferTarget();
   309→  if (opts.fileHandle) {
   310→    try {
   311→      writable = await opts.fileHandle.createWritable();
   312→      target = new FileSystemWritableFileStreamTarget(writable);
   313→    } catch {
   314→      writable = null; // gagal buka file → jatuh ke blob
   315→    }
   316→  }
   317→  if (!writable) target = new ArrayBufferTarget();
   318→  const muxer = new Muxer({
   319→    target,
   320→    video: { codec: "avc", width: dims.w, height: dims.h, frameRate: fps },
   321→    ...(audioSetup
   322→      ? {
   323→          audio: {
   324→            codec: audioSetup.muxerCodec,
   325→            sampleRate: audioSetup.cfg.sampleRate,
   326→            numberOfChannels: audioSetup.cfg.numberOfChannels,
   327→          },
   328→        }
   329→      : {}),
   330→    fastStart: writable ? false : ("in-memory" as const),
   331→  });
   332→
   333→  /* ----- encoder video ----- */
   334→  let encodeErr: Error | null = null;
   335→  const venc = new VideoEncoder({
   336→    output: (chunk, meta) => {
   337→      try {
   338→        muxer.addVideoChunk(chunk, meta);
   339→      } catch (e) {
   340→        encodeErr = e instanceof Error ? e : new Error(String(e));
   341→      }
   342→    },
   343→    error: (e) => {
   344→      encodeErr = e instanceof Error ? e : new Error(String(e));
   345→    },
   346→  });
   347→  venc.configure(videoCfg);
   348→
   349→  /* ----- encoder audio ----- */
   350→  let aenc: AudioEncoder | null = null;
   351→  if (audioSetup && mixed) {
   352→    aenc = new AudioEncoder({
   353→      output: (chunk, meta) => {
   354→        try {
   355→          muxer.addAudioChunk(chunk, meta);
   356→        } catch (e) {
   357→          encodeErr = e instanceof Error ? e : new Error(String(e));
   358→        }
   359→      },
   360→      error: (e) => {
   361→        encodeErr = e instanceof Error ? e : new Error(String(e));
   362→      },
   363→    });
   364→    aenc.configure(audioSetup.cfg);
   365→  }
   366→
   367→  /* ----- render frame per frame ----- */
   368→  const canvas = makeCanvas(dims.w, dims.h);
   369→  // v1.2.6 — openVideo: selalu ada batas waktu + pesan jelas
   370→  const v = await openVideo(media.url, 30000);
   371→
   372→  // siapkan keyframe zoom pembicara bila aktif
   373→  let keys: CropKey[] | null = null;
   374→  if (clip.speaker.enabled) {
   375→    opts.onProgress?.(0.08, "Analisis zoom pembicara…");
   376→    const analysis = await getCachedSpeaker(media, clip.speaker);
   377→    keys = buildCropKeys(analysis, clip.speaker, (media.width || 16) / (media.height || 9));
   378→  }
   379→
   380→  const totalFrames = Math.ceil(totalDur * fps);
   381→  let done = 0;
   382→  let outTs = 0; // timestamp output mikrodetik
   383→
   384→  for (const range of opts.ranges) {
   385→    const nFrames = Math.ceil((range.end - range.start) * fps);
   386→    for (let i = 0; i < nFrames; i++) {
   387→      if (opts.shouldCancel?.()) {
   388→        try { venc.close(); } catch { /* aman */ }
   389→        if (aenc) { try { aenc.close(); } catch { /* aman */ } }
   390→        throw new Error("Dibatalkan");
   391→      }
   392→      if (encodeErr) throw encodeErr;
   393→      if (venc.state !== "configured") {
   394→        throw new Error("Encoder video berhenti di tengah proses (driver/GPU) — coba ENCODER → CPU");
   395→      }
   396→
   397→      const t = range.start + i / fps;
   398→      // v1.2.6 — seek dilindungi batas waktu 45 dtk + tombol BATAL tetap hidup.
   399→      // Dulu: seek menggantung selamanya (rAF mati saat window minimize) →
   400→      // job beku di frame terakhir dan seluruh antrean mati.
   401→      await raceStall(
   402→        seekVideo(v, t),
   403→        opts.shouldCancel,
   404→        45_000,
   405→        new Error(
   406→          `Seek frame ${done + 1}/${totalFrames} (t=${t.toFixed(2)} dtk) macet >45 dtk — job dihentikan otomatis`
   407→        )
   408→      );
   409→      drawComposite({
   410→        src: v,
   411→        srcW: v.videoWidth || media.width,
   412→        srcH: v.videoHeight || media.height,
   413→        out: canvas,
   414→        t,
   415→        aspect: settings.aspect,
   416→        zoomPan: clip.zoomPan,
   417→        speakerKeys: keys,
   418→        speakerCfg: clip.speaker,
   419→        subtitle: clip.subtitle,
   420→        texts: clip.texts,
   421→        tracker: clip.tracker,
   422→        watermark: clip.watermark,
   423→        effects: clip.effects,
   424→      });
   425→      const frame = new VideoFrame(canvas, {
   426→        timestamp: Math.round(outTs),
   427→        duration: frameDurUs,
   428→      });
   429→      venc.encode(frame, { keyFrame: done % (fps * 2) === 0 });
   430→      frame.close();
   431→      outTs += frameDurUs;
   432→      done++;
   433→
   434→      if (done % 5 === 0 || done === totalFrames) {
   435→        opts.onProgress?.(
   436→          0.1 + (done / totalFrames) * 0.82,
   437→          `Encoding video ${done}/${totalFrames} frame · ${dims.w}×${dims.h}`
   438→        );
   439→        // beri napas UI & flush antrean agar mem stabil
   440→        // v1.2.6 — tunggu antrean dibatasi 30 dtk + cek error encoder tiap putaran
   441→        if (venc.encodeQueueSize > 12) {
   442→          await new Promise((r) => setTimeout(r, 0));
   443→          const t0 = Date.now();
   444→          while (venc.encodeQueueSize > 6) {
   445→            if (encodeErr) throw encodeErr;
   446→            if (venc.state !== "configured") {
   447→              throw new Error("Encoder video berhenti (antrean tidak berkurang) — coba ENCODER → CPU");
   448→            }
   449→            if (opts.shouldCancel?.()) throw new Error("Dibatalkan");
   450→            if (Date.now() - t0 > 30_000) {
   451→              throw new Error("Encoder video macet (antrean encode tidak berkurang 30 dtk) — job dihentikan otomatis");
   452→            }
   453→            await new Promise((r) => setTimeout(r, 8));
   454→          }
   455→        } else {
   456→          await new Promise((r) => setTimeout(r, 0));
   457→        }
   458→      }
   459→    }
   460→  }
   461→
   462→  opts.onProgress?.(0.94, "Finalisasi video…");
   463→  await venc.flush();
   464→  venc.close();
   465→
   466→  /* ----- audio chunks ----- */
   467→  if (aenc && mixed) {
   468→    opts.onProgress?.(0.96, "Encoding audio AAC…");
   469→    const sr = mixed.sampleRate;
   470→    const chunkFrames = 4800; // 0.1s @48k
   471→    const chans = Math.min(2, mixed.numberOfChannels);
   472→    for (let off = 0; off < mixed.length; off += chunkFrames) {
   473→      const n = Math.min(chunkFrames, mixed.length - off);
   474→      const data = new Float32Array(n * chans);
   475→      for (let c = 0; c < chans; c++) {
   476→        const src = mixed.getChannelData(Math.min(c, mixed.numberOfChannels - 1));
   477→        data.set(src.subarray(off, off + n), c * n);
   478→      }
   479→      const ad = new AudioData({
   480→        format: "f32-planar",
   481→        sampleRate: sr,
   482→        numberOfFrames: n,
   483→        numberOfChannels: chans,
   484→        timestamp: Math.round((off / sr) * 1e6),
   485→        data,
   486→      });
   487→      aenc.encode(ad);
   488→      ad.close();
   489→      // v1.2.6 — antrean audio benar-benar dikuras (dulu cuma sleep 4 ms
   490→      // tanpa menunggu → bisa membengkak di video panjang)
   491→      if (aenc.encodeQueueSize > 16) {
   492→        const t0 = Date.now();
   493→        while (aenc.encodeQueueSize > 8) {
   494→          if (encodeErr) throw encodeErr;
   495→          if (Date.now() - t0 > 30_000) throw new Error("Encoder audio macet — job dihentikan otomatis");
   496→          await new Promise((r) => setTimeout(r, 4));
   497→        }
   498→      }
   499→    }
   500→    await aenc.flush();
   501→    aenc.close();
   502→  }
   503→
   504→  if (encodeErr) throw encodeErr;
   505→
   506→  muxer.finalize();
   507→  opts.onProgress?.(1, "Selesai");
   508→  v.removeAttribute("src");
   509→  v.load();
   510→  canvas.width = 0;
   511→  // v1.2.7 — jalur file: data sudah tertulis di disk selagi encode;
   512→  // close() menyelesaikan file. Tidak ada blob, tidak ada unduhan.
   513→  if (writable) {
   514→    try { await writable.close(); } catch { /* file sudah tertulis */ }
   515→    return null;
   516→  }
   517→  const blob = new Blob([(target as ArrayBufferTarget).buffer], { type: "video/mp4" });
   518→  return blob;
   519→}

   531→/* ---------- pemilih encoder audio (AAC → Opus fallback) ---------- */
   532→
   533→interface AudioSetup {
   534→  cfg: AudioEncoderConfig;
   535→  muxerCodec: "aac" | "opus";
   536→}

   538→async function pickAudioConfig(preferSampleRate = 48000): Promise<AudioSetup | null> {
   539→  const bitrate = 160_000;
   540→  const candidates: AudioEncoderConfig[] = [
   541→    { codec: "mp4a.40.2", sampleRate: preferSampleRate, numberOfChannels: 2, bitrate },
   542→    { codec: "mp4a.40.2", sampleRate: 44100, numberOfChannels: 2, bitrate },
   543→    { codec: "mp4a.40.2", sampleRate: 44100, numberOfChannels: 1, bitrate },
   544→    { codec: "opus", sampleRate: 48000, numberOfChannels: 2, bitrate },
   545→  ];
   546→  for (const c of candidates) {
   547→    try {
   548→      if (!("AudioEncoder" in window)) return null;
   549→      const sup = await AudioEncoder.isConfigSupported(c);
   550→      if (sup.supported) {
   551→        return {
   552→          cfg: (sup.config as AudioEncoderConfig) ?? c,
   553→          muxerCodec: c.codec === "opus" ? "opus" : "aac",
   554→        };
   555→      }
   556→    } catch {
   557→      /* coba kandidat berikutnya */
   558→    }
   559→  }
   560→  return null;
   561→}

   563→async function runJob(job: ExportJob, media: MediaItem[], clip: ClipSettings, update: (p: Partial<ExportJob>) => void) {
   564→  const rangesAll = job.ranges;
   565→  try {
   566→    update({ status: "running", stage: "Mulai…", progress: 0, startedAt: Date.now() });
   567→
   568→    if (job.target === "split") {
   569→      // ekspor tiap bagian sebagai file terpisah
   570→      const parts = rangesAll;
   571→      /* v1.2.6 — audio 48 kHz dibaca SEKALI untuk semua bagian.
   572→         Dulu: tiap bagian melakukan fetch + decode penuh berulang
   573→         (20 bagian = 20× decode video panjang) → lambat & RAM meledak. */
   574→      let audioPre: AudioBuffer | null = null;
   575→      if (media[0].type === "video") {
   576→        update({ stage: `Membaca audio (sekali untuk ${parts.length} bagian)…`, progress: 0.004 });
   577→        try {
   578→          audioPre = await decodeAudioFile(media[0].url);
   579→        } catch {
   580→          audioPre = null; // tanpa audio / codec aneh → lanjut tanpa audio
   581→        }
   582→      }
   583→      let ok = 0;
   584→      // v1.2.7 — folder simpanan diminta SEKALI di awal (bila mode FOLDER)
   585→      const useDir = getSaveMode() === "folder" ? await obtainSaveDir(false) : null;
   586→      for (let i = 0; i < parts.length; i++) {
   587→        if (cancelFlags.has(job.id)) throw new Error("Dibatalkan");
   588→        const r = parts[i];
   589→        const partName = `${safeName(job.name)}_BAGIAN_${String(i + 1).padStart(2, "0")}.mp4`;
   590→        const fh = useDir ? await resolveSaveFile(partName) : null;
   591→        const blob = await exportOne({
   592→          media: media[0],
   593→          ranges: [{ start: r.start, end: r.end }],
   594→          clip: clip,
   595→          name: `${job.name}_part${String(i + 1).padStart(2, "0")}`,
   596→          audioPre,
   597→          fileHandle: fh,
   598→          onProgress: (p, stage) =>
   599→            update({ progress: (i + p) / parts.length * 0.99, stage: `Bagian ${i + 1}/${parts.length} — ${stage}` }),
   600→          shouldCancel: () => cancelFlags.has(job.id),
   601→        });
   602→        if (blob) {
   603→          /* v1.2.6 — tiap bagian DIUNDUH SEGERA setelah jadi.
   604→             Dulu semua blob ditumpuk di RAM sampai bagian terakhir
   605→             (20 × ratusan MB) → renderer OOM crash di tengah proses. */
   606→          await assertBlobOk(blob, `Bagian ${i + 1}`);
   607→          downloadBlob(blob, partName);
   608→        }
   609→        ok++;
   610→        await new Promise((res) => setTimeout(res, 600));
   611→      }
   612→      update({ status: "done", progress: 1, stage: `${ok} file${useDir ? ` tersimpan di "${useDir.name}"` : " terunduh"}`, finishedAt: Date.now() });
   613→      return;
   614→    }
   615→
   616→    if (job.target === "batch") {
   617→      /* v1.2.6 — BUG LAMA: batch memakai media[0] untuk SEMUA range →
   618→         video ke-2 dst ter-export SALAH (frame video pertama di-clamp).
   619→         Kini tiap video diproses sendiri + diunduh segera. */
   620→      let ok = 0;
   621→      const gagal: string[] = [];
   622→      // v1.2.7 — folder simpanan diminta SEKALI di awal (bila mode FOLDER)
   623→      const useDirB = getSaveMode() === "folder" ? await obtainSaveDir(false) : null;
   624→      for (let i = 0; i < rangesAll.length; i++) {
   625→        if (cancelFlags.has(job.id)) throw new Error("Dibatalkan");
   626→        const r = rangesAll[i];
   627→        const m = media[i] ?? media[0];
   628→        try {
   629→          const outName = `${safeName(m.name)}_${String(i + 1).padStart(2, "0")}.mp4`;
   630→          const fh = useDirB ? await resolveSaveFile(outName) : null;
   631→          const blob = await exportOne({
   632→            media: m,
   633→            ranges: [{ start: r.start, end: r.end }],
   634→            clip: { ...clip },
   635→            name: `${job.name}_video${String(i + 1).padStart(2, "0")}`,
   636→            fileHandle: fh,
   637→            onProgress: (p, stage) =>
   638→              update({
   639→                progress: (i + p) / rangesAll.length * 0.99,
   640→                stage: `Video ${i + 1}/${rangesAll.length} (${m.name.slice(0, 24)}) — ${stage}`,
   641→              }),
   642→            shouldCancel: () => cancelFlags.has(job.id),
   643→          });
   644→          if (blob) {
   645→            await assertBlobOk(blob, m.name.slice(0, 24));
   646→            downloadBlob(blob, outName);
   647→          }
   648→          ok++;
   649→          await new Promise((res) => setTimeout(res, 600));
   650→        } catch (e) {
   651→          const msg = e instanceof Error ? e.message : String(e);
   652→          if (msg === "Dibatalkan") throw e;
   653→          gagal.push(`${m.name.slice(0, 24)}: ${msg.slice(0, 80)}`);
   654→        }
   655→      }
   656→      if (!ok) throw new Error(gagal.join(" · ") || "Semua video gagal");
   657→      update({
   658→        status: "done", progress: 1, finishedAt: Date.now(),
   659→        stage: `${ok}/${rangesAll.length} video${useDirB ? ` tersimpan di "${useDirB.name}"` : " terunduh"}${gagal.length ? ` · ${gagal.length} gagal` : ""}`,
   660→      });
   661→      return;
   662→    }
   663→
   664→    // current / range — satu file output dari satu sumber
   665→    if (job.target !== "merge" || media.length === 1) {
   666→      // v1.2.7 — SIMPAN KE FOLDER: MP4 ditulis langsung ke disk selagi
   667→      // encode (tanpa blob besar di RAM). Gagal pilih folder → jalur
   668→      // unduhan browser + pengeccek kesehatan blob (bukan 0 byte diam).
   669→      const outName = `${safeName(job.name)}.mp4`;
   670→      const fh = getSaveMode() === "folder" ? await resolveSaveFile(outName) : null;
   671→      const blob = await exportOne({
   672→        media: media[0],
   673→        ranges: rangesAll.map((r) => ({ start: r.start, end: r.end })),
   674→        clip: { ...clip },
   675→        name: job.name,
   676→        fileHandle: fh,
   677→        onProgress: (p, stage) => update({ progress: p * 0.99, stage }),
   678→        shouldCancel: () => cancelFlags.has(job.id),
   679→      });
   680→      if (blob) {
   681→        await assertBlobOk(blob, "Ekspor video");
   682→        downloadBlob(blob, outName);
   683→        update({ status: "done", progress: 1, stage: "MP4 terunduh", finishedAt: Date.now() });
   684→      } else {
   685→        update({ status: "done", progress: 1, stage: `MP4 tersimpan: ${outName}`, finishedAt: Date.now() });
   686→      }
   687→      return;
   688→    }
   689→
   690→    // merge multi-sumber
   691→    const mergeName = `${safeName(job.name)}.mp4`;
   692→    const mergeFh = getSaveMode() === "folder" ? await resolveSaveFile(mergeName) : null;
   693→    const blob = await exportMerge(
   694→      media.map((m, i) => ({ media: m, ranges: [rangesAll[i]] })),
   695→      clip,
   696→      job.name,
   697→      (p, stage) => update({ progress: p * 0.99, stage }),
   698→      () => cancelFlags.has(job.id),
   699→      mergeFh
   700→    );
   701→    if (blob) {
   702→      await assertBlobOk(blob, "Gabung video");
   703→      downloadBlob(blob, mergeName);
   704→      update({ status: "done", progress: 1, stage: "Gabungan MP4 terunduh", finishedAt: Date.now() });
   705→    } else {
   706→      update({ status: "done", progress: 1, stage: `Gabungan tersimpan: ${mergeName}`, finishedAt: Date.now() });
   707→    }
   708→  } catch (e) {
   709→    const msg = e instanceof Error ? e.message : String(e);
   710→    update({
   711→      status: msg === "Dibatalkan" ? "canceled" : "error",
   712→      error: msg,
   713→      stage: msg === "Dibatalkan" ? "Dibatalkan" : "Gagal",
   714→      finishedAt: Date.now(),
   715→    });
   716→  } finally {
   717→    cancelFlags.delete(job.id);
   718→  }
   719→}

   721→/** GABUNG VIDEO — banyak sumber menjadi satu MP4 (hard cut) */
   722→export async function exportMerge(
   723→  items: { media: MediaItem; ranges: { start: number; end: number }[] }[],
   724→  clip: ClipSettings,
   725→  name: string,
   726→  onProgress?: (p: number, stage: string) => void,
   727→  shouldCancel?: () => boolean,
   728→  fileHandle?: FileSystemFileHandle | null
   729→): Promise<Blob | null> {
   730→  // gabung jadi satu daftar range lintas sumber; exportOne memproses berurutan
   731→  // implementasi: render per sumber ke buffer frame → terlalu berat; gunakan
   732→  // strategi encode berurutan dengan muxer bersama via exportMulti
   733→  return exportMulti(items.map((it) => ({ media: it.media, ranges: it.ranges })), clip, onProgress, shouldCancel, fileHandle);
   734→}

   736→/** ekspor multi-sumber ke SATU file (dipakai gabung video) */
   737→async function exportMulti(
   738→  sources: { media: MediaItem; ranges: { start: number; end: number }[] }[],
   739→  clip: ClipSettings,
   740→  onProgress?: (p: number, stage: string) => void,
   741→  shouldCancel?: () => boolean,
   742→  fileHandle?: FileSystemFileHandle | null
   743→): Promise<Blob | null> {
   744→  const settings = clip.export;
   745→  const first = sources[0].media;
   746→  const totalDur = sources.reduce((a, s) => a + s.ranges.reduce((b, r) => b + r.end - r.start, 0), 0);
   747→  await ensureFontsLoaded();
   748→
   749→  const dims = outputDims(settings, first.width || 1920, first.height || 1080);
   750→  const bitrate = videoBitrate(settings, totalDur, first.width || 1920, first.height || 1080);
   751→  const codec = h264Codec(settings, dims.w, dims.h);
   752→  const fps = settings.fps;
   753→  const frameDurUs = Math.round(1e6 / fps);
   754→
   755→  const audioSetup = await pickAudioConfig(48000);
   756→  // v1.2.7 — dukung tulis langsung ke disk (sama seperti exportOne)
   757→  let writable: FileSystemWritableFileStream | null = null;
   758→  let target: ArrayBufferTarget | FileSystemWritableFileStreamTarget = new ArrayBufferTarget();
   759→  if (fileHandle) {
   760→    try {
   761→      writable = await fileHandle.createWritable();
   762→      target = new FileSystemWritableFileStreamTarget(writable);
   763→    } catch {
   764→      writable = null;
   765→    }
   766→  }
   767→  if (!writable) target = new ArrayBufferTarget();
   768→  const muxer = new Muxer({
   769→    target,
   770→    video: { codec: "avc", width: dims.w, height: dims.h, frameRate: fps },
   771→    ...(audioSetup
   772→      ? {
   773→          audio: {
   774→            codec: audioSetup.muxerCodec,
   775→            sampleRate: audioSetup.cfg.sampleRate,
   776→            numberOfChannels: audioSetup.cfg.numberOfChannels,
   777→          },
   778→        }
   779→      : {}),
   780→    fastStart: writable ? false : ("in-memory" as const),
   781→  });
   782→
   783→  let encodeErr: Error | null = null;
   784→  const venc = new VideoEncoder({
   785→    output: (chunk, meta) => muxer.addVideoChunk(chunk, meta),
   786→    error: (e) => (encodeErr = e instanceof Error ? e : new Error(String(e))),
   787→  });
   788→  venc.configure({
   789→    codec, width: dims.w, height: dims.h, bitrate, framerate: fps,
   790→    latencyMode: "quality",
   791→  } as VideoEncoderConfig);
   792→
   793→  const aenc = new AudioEncoder({
   794→    output: (chunk, meta) => muxer.addAudioChunk(chunk, meta),
   795→    error: (e) => (encodeErr = e instanceof Error ? e : new Error(String(e))),
   796→  });
   797→  if (audioSetup) aenc.configure(audioSetup.cfg);
   798→
   799→  const canvas = makeCanvas(dims.w, dims.h);
   800→  const totalFrames = Math.ceil(totalDur * fps);
   801→  let done = 0;
   802→  let outTs = 0;
   803→  /* v1.2.6 — timestamp audio GLOBAL: dulu mulai dari 0 per sumber,
   804→     sehingga audio video ke-2 menimpa audio video pertama di muxer
   805→     (output rusak/saling tumpang). Kini offset berjalan sepanjang gabungan. */
   806→  let audioTsUs = 0;
   807→
   808→  for (const src of sources) {
   809→    // v1.2.6 — openVideo: batas waktu + pesan jelas per sumber
   810→    const v = await openVideo(src.media.url, 30000).catch((e) => {
   811→      throw new Error(`${src.media.name}: ${e instanceof Error ? e.message : e}`);
   812→    });
   813→
   814→    let keys: CropKey[] | null = null;
   815→    if (clip.speaker.enabled) {
   816→      const analysis = await getCachedSpeaker(src.media, clip.speaker);
   817→      keys = buildCropKeys(analysis, clip.speaker, (src.media.width || 16) / (src.media.height || 9));
   818→    }
   819→
   820→    // audio sumber → decode sekali per sumber
   821→    let origBuf: AudioBuffer | null = null;
   822→    try {
   823→      origBuf = await decodeAudioFile(src.media.url);
   824→    } catch { origBuf = null; }
   825→
   826→    let musicBuf: AudioBuffer | null = null;
   827→    if (clip.music.preset !== "none") {
   828→      musicBuf = clip.music.customBuffer ?? (await composeMusic(clip.music.preset, totalDur, clip.music.fadeOut));
   829→    }
   830→
   831→    const segDur = src.ranges.reduce((a, r) => a + r.end - r.start, 0);
   832→    let mixedSeg: AudioBuffer | null = null;
   833→    if (audioSetup && (origBuf || musicBuf)) {
   834→      mixedSeg = await renderMix({
   835→        duration: segDur,
   836→        sampleRate: audioSetup.cfg.sampleRate,
   837→        channels: audioSetup.cfg.numberOfChannels,
   838→        original: origBuf,
   839→        originalStart: src.ranges[0]?.start ?? 0,
   840→        originalGain: musicBuf ? 1 - clip.music.duckOriginal * 0.8 : 1,
   841→        music: musicBuf,
   842→        musicGain: clip.music.gain,
   843→        musicLoop: true,
   844→        fadeIn: 0.3,
   845→        fadeOut: 0.3,
   846→      });
   847→    }
   848→
   849→    for (const range of src.ranges) {
   850→      const nFrames = Math.ceil((range.end - range.start) * fps);
   851→      for (let i = 0; i < nFrames; i++) {
   852→        if (shouldCancel?.()) {
   853→          try { venc.close(); aenc.close(); } catch { /* aman */ }
   854→          throw new Error("Dibatalkan");
   855→        }
   856→        if (encodeErr) throw encodeErr;
   857→        if (venc.state !== "configured") {
   858→          throw new Error("Encoder video berhenti di tengah proses — coba ENCODER → CPU");
   859→        }
   860→        const t = range.start + i / fps;
   861→        // v1.2.6 — seek dilindungi batas waktu + BATAL tetap bereaksi
   862→        await raceStall(
   863→          seekVideo(v, t),
   864→          shouldCancel,
   865→          45_000,
   866→          new Error(`Seek macet >45 dtk (t=${t.toFixed(2)} dtk) — job dihentikan otomatis`)
   867→        );
   868→        drawComposite({
   869→          src: v,
   870→          srcW: v.videoWidth || src.media.width,
   871→          srcH: v.videoHeight || src.media.height,
   872→          out: canvas,
   873→          t,
   874→          aspect: settings.aspect,
   875→          zoomPan: clip.zoomPan,
   876→          speakerKeys: keys,
   877→          speakerCfg: clip.speaker,
   878→          subtitle: clip.subtitle,
   879→          texts: clip.texts,
   880→          tracker: clip.tracker,
   881→          watermark: clip.watermark,
   882→          effects: clip.effects,
   883→        });
   884→        const frame = new VideoFrame(canvas, { timestamp: Math.round(outTs), duration: frameDurUs });
   885→        venc.encode(frame, { keyFrame: done % (fps * 2) === 0 });
   886→        frame.close();
   887→        outTs += frameDurUs;
   888→        done++;
   889→        if (done % 5 === 0 || done === totalFrames) {
   890→          onProgress?.(0.05 + (done / totalFrames) * 0.9, `Gabung ${done}/${totalFrames} frame`);
   891→          await new Promise((r) => setTimeout(r, 0));
   892→        }
   893→      }
   894→    }
   895→
   896→    // audio segmen ini — timestamp GLOBAL (v1.2.6)
   897→    if (mixedSeg && audioSetup) {
   898→      const sr = mixedSeg.sampleRate;
   899→      const chunkFrames = 4800;
   900→      const chans = Math.min(2, mixedSeg.numberOfChannels);
   901→      for (let off = 0; off < mixedSeg.length; off += chunkFrames) {
   902→        const n = Math.min(chunkFrames, mixedSeg.length - off);
   903→        const data = new Float32Array(n * chans);
   904→        for (let c = 0; c < chans; c++) {
   905→          const s = mixedSeg.getChannelData(Math.min(c, mixedSeg.numberOfChannels - 1));
   906→          data.set(s.subarray(off, off + n), c * n);
   907→        }
   908→        const ad = new AudioData({
   909→          format: "f32-planar", sampleRate: sr, numberOfFrames: n,
   910→          numberOfChannels: chans,
   911→          timestamp: Math.round(audioTsUs + (off / sr) * 1e6), data,
   912→        });
   913→        aenc.encode(ad);
   914→        ad.close();
   915→        if (aenc.encodeQueueSize > 16) {
   916→          const t0 = Date.now();
   917→          while (aenc.encodeQueueSize > 8) {
   918→            if (encodeErr) throw encodeErr;
   919→            if (Date.now() - t0 > 30_000) throw new Error("Encoder audio macet — job dihentikan otomatis");
   920→            await new Promise((r) => setTimeout(r, 4));
   921→          }
   922→        }
   923→      }
   924→      audioTsUs += Math.round((mixedSeg.length / sr) * 1e6);
   925→    }
   926→
   927→    v.removeAttribute("src");
   928→    v.load();
   929→  }
   930→
   931→  onProgress?.(0.96, "Finalisasi…");
   932→  await venc.flush();
   933→  venc.close();
   934→  if (audioSetup) {
   935→    await aenc.flush();
   936→    aenc.close();
   937→  }
   938→  if (encodeErr) throw encodeErr;
   939→  muxer.finalize();
   940→  canvas.width = 0;
   941→  // v1.2.7 — jalur file: sudah tertulis di disk; jalur blob: bungkus buffer
   942→  if (writable) {
   943→    try { await writable.close(); } catch { /* file sudah tertulis */ }
   944→    return null;
   945→  }
   946→  return new Blob([(target as ArrayBufferTarget).buffer], { type: "video/mp4" });
   947→}

   951→export async function processJob(
   952→  job: ExportJob,
   953→  getMedia: (id: string) => MediaItem | undefined,
   954→  clip: ClipSettings,
   955→  update: (p: Partial<ExportJob>) => void
   956→) {
   957→  const media = job.ranges.map((r) => getMedia(r.mediaId)).filter(Boolean) as MediaItem[];
   958→  if (!media.length) {
   959→    update({ status: "error", error: "Media tidak ditemukan" });
   960→    return;
   961→  }
   962→  await runJob(job, media, clip, update);
   963→}
